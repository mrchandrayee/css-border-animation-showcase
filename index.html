Rotating Border

Rotating Border CSS

@property --angle {
  syntax: '<angle>';
  initial-value: 0deg;
  inherits: false;
}
.rotating::before {
  background: conic-gradient(
    from var(--angle),
    #22c55e, #3b82f6, #a855f7,
    #f59e0b, #22c55e
  );
  animation: spin 3s linear infinite;
}
@keyframes spin {
  to { --angle: 360deg; }
}
